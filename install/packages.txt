# dotfiles-Fedora/install/packages.txt
# dnf package names for the modern terminal stack. One per line; # comments ok.
# starship, atuin and yazi are installed by bootstrap.sh instead — keep them OUT
# of here. Two different reasons, so don't collapse them back into one line:
#   starship, yazi — genuinely absent from every current Fedora release; the
#     upstream installer (starship) / cargo (yazi) is the only automatic path.
#   atuin — IS packaged now (F43/F44/rawhide all carry 18.12.1), but Fedora trails
#     upstream 18.19.0, and 18.19.0 is the exact client core/PORTING-MATRIX.md's
#     atuin²⁰ note validated the daemon config-precedence behavior against. Taking
#     the RPM would swap a known-good atuin for an unvalidated one, so the upstream
#     installer stays. Revisit when Fedora catches up: bootstrap.sh's three atuin
#     blocks are all presence-guarded, so adding the name here is the only edit
#     it would take.

# ── shell + multiplexer ───────────────────────────────────────
zsh
tmux

# ── modern CLI core (all in Fedora repos as of F40+) ──────────
eza            # ls
bat            # cat
fd-find        # find  (binary: fd on Fedora, not fdfind)
ripgrep        # grep
zoxide         # cd
fzf            # fuzzy finder
git-delta      # git pager / diffs
btop           # top/htop
tealdeer       # tldr (fast Rust client; binary: tldr) — ORPHANED, last built F44; no
               # rawhide/F45 build. Kept here for F43/F44; bootstrap.sh cargo-installs
               # it wherever dnf no longer resolves the name.

# ── modern CLI extras (core-doctor probes these; all in Fedora repos) ─
duf            # df   (disk usage/free)
procs          # ps — ORPHANED, last built F44; no rawhide/F45 build. Same deal as
               # tealdeer above: kept for F43/F44, cargo fallback covers F45+.
yq             # jq-for-YAML — Fedora's `yq` IS mikefarah's Go build (python-yq is the separate Python tool)
glow           # markdown viewer
gum            # shell-script UI toolkit (charm)
# dust, xh, sd, doggo, carapace, sesh, gron, op (1Password CLI): NOT reliably in
# Fedora repos — bootstrap.sh installs them via cargo / go / the 1Password repo.
# Two were dropped by Fedora rather than never packaged: gron was orphaned (last
# shipped F41/42) and sd was retired after F41 (rust-sd's `sd` subpackage, last
# built 1.0.0-4.fc41), so `dnf install gron` / `dnf install sd` both fail on a
# current release — hence the go/cargo fallbacks.
# tealdeer and procs are mid-way down that same road: both went orphan and neither
# was rebuilt for rawhide/F45, but both still install on F43/F44. They therefore stay
# ACTIVE in this list (removing them would swap a fast RPM for a slow source build on
# every currently-supported release) and bootstrap.sh carries presence-guarded cargo
# fallbacks for them. The dnf install uses --skip-unavailable, so once F45 drops the
# names the transaction skips them instead of failing.

neovim

openssh-clients  # ssh / ssh-keygen / ssh-agent — Workstation has it; minimal/Server may not

# ── build + language toolchain (needed for cargo-installed tools) ─

cargo          # rust toolchain -> backstops every `cargo install` in bootstrap.sh, and

               # don't list the dependents here: bootstrap.sh is the source of truth for
               # which tools fall back to cargo, so an inventory in this file is just a
               # second copy to keep in sync.
